// .html()
// Gets (or sets) everything INSIDE the element, including HTML tags.

<div id="fruit"><b>Apple</b></div>

$("#fruit").html();      // "<b>Apple</b>"
$("#fruit").html("Banana"); // Changes the inside to: Banana
---------------------------------------------------------
// .text()
// Gets (or sets) only the visible text.
// HTML tags are ignored.

<div id="fruit"><b>Apple</b></div>

$("#fruit").text();      // "Apple"
$("#fruit").text("Banana"); // Changes the text to: Banana

<div id="duck">minute<b>10m</b></div> $("#duck").text(); // "minute10m"

.text() returns all visible text, even the text inside child elements.
---------------------------------------------------------
// .attr()
// Gets* (or sets) the value of ANY HTML attribute.

<div id="fruit" value="10" color="red"></div>

$("#fruit").attr("value"); // "10"
$("#fruit").attr("color"); // "red"
$("#fruit").attr("id");    // "fruit"

// And could Set* an attribute
$("#fruit").attr("value", "20");
// Result: <div id="fruit" value="20" color="red"></div>
---------------------------------------------------------
// .val()
// Gets (or sets) the value of FORM elements.
// Usually used with:
// <input>
// <textarea>
// <select>
// <option>
// <button>

<input id="age" value="28">

$("#age").val();      // "28"
$("#age").val("29");  // Changes the value to 29
